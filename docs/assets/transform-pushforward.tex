% title: Pushforward adds input and output changes
% description: The original IR maps inputs to an output. Pushforward adds an input-change channel and an output-change channel. Both channels run forward.
\documentclass[tikz,border=5pt]{standalone}
\input{parts/transform-style.tex}
\begin{document}
\begin{tikzpicture}[transform panel]
    \transformcanvas

    \node[transform program, minimum height=1.0cm] (original) at (0,1.5) {Original IR};
    \node (inputs) at (-5,1.5) {Inputs};
    \node (output) at (5,1.5) {Output};
    \draw[flow] (inputs.east) -- (original.west);
    \draw[flow] (original.east) -- (output.west);

    \node[transform program, minimum height=1.65cm] (new) at (0,-1.35) {Forward IR};
    \transformarrow[right]{original.south}{new.north}{\texttt{pushforward}}
    \node (newin) at (-5,-0.9) {Inputs};
    \node (changein) at (-5,-1.8) {Input changes};
    \node (newout) at (5,-0.9) {Output};
    \node (changeout) at (5,-1.8) {Output change};
    \draw[flow] (newin.east) -- ([yshift=0.45cm]new.west);
    \draw[flow] ([yshift=0.45cm]new.east) -- (newout.west);
    \draw[flow] (changein.east) -- ([yshift=-0.45cm]new.west);
    \draw[flow] ([yshift=-0.45cm]new.east) -- (changeout.west);

\end{tikzpicture}
\end{document}
