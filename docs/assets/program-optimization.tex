    \node[node] (method) at (8.4, 0) {Optimization\\method};

    \draw[flow] (spaces.east) -- (transforms.west);
    \draw[flow] (rules.east) -- (transforms.west);
    \draw[flow] (transforms.east) -- node[above, font=\small] {supports} (method.west);
